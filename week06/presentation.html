수업 진행 방식

  1. 백엔드 API가 먼저 되는지 확인합니다.

CorsConfig.java (2/2)

            @Override
            public void addCorsMappings(CorsRegistry registry) {
                registry.addMapping("/api/**")
                        .allowedOrigins("http://localhost:3000")
                        .allowedMethods("GET", "POST", "PUT", "PATCH", "DELETE", "OPTIONS")
                        .allowedHeaders("*");
            }
        };
    }
}

React 개발 서버 포트가 5173이면 allowedOrigins("http://localhost:5173")도 추가해야 합니다.

Lab 4. App.js 전체 구현

처음에는 Hook 분리 없이 App.js 한 파일로 완성합니다.

이유:

  • API가 실제로 연결되는지 먼저 확인한다.
  • 상태, 로딩, 에러 흐름을 한 화면에서 본다.
  • 동작한 뒤 Hook으로 분리하면 이해가 쉽다.

App.js (1/6) import와 state

import { useEffect, useState } from 'react';
import api from './api/axios';

function App() {
  const [items, setItems] = useState([]);
  const [title, setTitle] = useState('');
  const [loading, setLoading] = useState(false);
  const [message, setMessage] = useState('');
  const [workingId, setWorkingId] = useState(null);

App.js (2/6) 목록 불러오기

  const load = async () => {
    setLoading(true);
    setMessage('');

    try {
      const res = await api.get('/todos');
      setItems(Array.isArray(res.data) ? res.data : []);
    } catch (e) {
      console.error(e);
      setMessage('목록 불러오기 실패 - 백엔드 실행과 CORS를 확인하세요.');
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    load();
  }, []);

App.js (3/6) 추가하기

  const handleAdd = async (e) => {
    e.preventDefault();

    if (!title.trim()) {
      setMessage('할 일을 입력하세요.');
      return;
    }

    try {
      await api.post('/todos', { title, description: '' });
      setTitle('');
      await load();
    } catch (e) {
      console.error(e);
      setMessage('추가 실패 - 서버 로그와 Network 탭을 확인하세요.');
    }
  };

App.js (4/6) 토글과 삭제

  const handleToggle = async (id) => {
    setWorkingId(id);
    try {
      await api.patch(`/todos/${id}/toggle`);
      await load();
    } catch (e) {
      console.error(e);
      setMessage('완료 상태 변경 실패');
    } finally {
      setWorkingId(null);
    }
  };

  const handleDelete = async (id) => {
    setWorkingId(id);
    try {
      await api.delete(`/todos/${id}`);
      await load();
    } catch (e) {
      console.error(e);
      setMessage('삭제 실패');
    } finally {
      setWorkingId(null);
    }
  };

App.js (5/6) 화면 상단

  return (
    <div style={{ maxWidth: 560, margin: '40px auto', fontFamily: 'sans-serif' }}>
      <h1>Todo 연동 실습</h1>

      <form onSubmit={handleAdd} style={{ marginBottom: 16 }}>
        <input
          value={title}
          onChange={(e) => setTitle(e.target.value)}
          placeholder="할 일 입력"
          style={{ width: '70%', padding: 8 }}
        />
        <button type="submit" style={{ padding: '8px 12px', marginLeft: 8 }}>
          추가
        </button>
      </form>

      {loading && <p>불러오는 중...</p>}
      {message && <p style={{ color: 'crimson' }}>{message}</p>}

App.js (6/6) 목록 출력

      <ul style={{ paddingLeft: 18 }}>
        {items.map((todo) => (
          <li key={todo.id} style={{ marginBottom: 8 }}>
            <button
              type="button"
              disabled={workingId === todo.id}
              onClick={() => handleToggle(todo.id)}
            >
              {todo.completed ? '완료' : '미완료'}
            </button>
            <span style={{ marginLeft: 8, textDecoration: todo.completed ? 'line-through' : 'none' }}>
              {todo.title}
            </span>
            <button type="button" onClick={() => handleDelete(todo.id)} style={{ marginLeft: 8 }}>
              삭제
            </button>
          </li>
        ))}
      </ul>
    </div>
  );
}

export default App;

자주 나는 문제 1: CORS

Access to XMLHttpRequest at 'http://localhost:8080/api/todos'
from origin 'http://localhost:3000' has been blocked by CORS policy

확인할 것:

  • Spring Boot에 CorsConfig 추가했는지
  • allowedOrigins("http://localhost:3000")인지
  • 백엔드를 재실행했는지

자주 나는 문제 2: 서버 연결 실패

ERR_CONNECTION_REFUSED
Network Error

확인할 것:

  • Spring Boot 서버가 켜져 있는지
  • 백엔드 포트가 8080이 맞는지
  • baseURL이 http://localhost:8080/api인지

자주 나는 문제 3: 404

GET http://localhost:8080/todos 404
GET http://localhost:8080/api/todos 200

대부분 /api 경로가 빠진 경우입니다.

const api = axios.create({

Lab 6. 커스텀 Hook으로 분리

App.js에서 API 호출 로직을 빼면 컴포넌트가 읽기 쉬워집니다.

src/hooks/useTodos.js

먼저 App.js 한 파일로 동작시킨 뒤 분리하는 순서를 추천합니다.

useTodos.js (1/4)

import { useCallback, useEffect, useState } from 'react';
import api from '../api/axios';

export default function useTodos() {
  const [todos, setTodos] = useState([]);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState(null);

useTodos.js (2/4)

  const fetchTodos = useCallback(async () => {
    setLoading(true);
    setError(null);

    try {
      const res = await api.get('/todos');
      setTodos(Array.isArray(res.data) ? res.data : []);
    } catch (e) {
      setError(e);
    } finally {
      setLoading(false);
    }
  }, []);

  useEffect(() => {
    fetchTodos();
  }, [fetchTodos]);

useTodos.js (3/4)

  const createTodo = async (payload) => {
    const res = await api.post('/todos', payload);
    await fetchTodos();
    return res.data;
  };

  const toggleTodo = async (id) => {
    const res = await api.patch(`/todos/${id}/toggle`);
    await fetchTodos();
    return res.data;
  };

  const deleteTodo = async (id) => {
    await api.delete(`/todos/${id}`);
    await fetchTodos();
  };

useTodos.js (4/4)

  return {
    todos,
    loading,
    error,
    fetchTodos,
    createTodo,
    toggleTodo,
    deleteTodo,
  };
}
